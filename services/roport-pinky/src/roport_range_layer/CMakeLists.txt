cmake_minimum_required(VERSION 3.8)
project(roport_range_layer)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(fresh_range_layer SHARED src/fresh_range_layer.cpp)
target_compile_features(fresh_range_layer PUBLIC cxx_std_17)
ament_target_dependencies(fresh_range_layer nav2_costmap_2d pluginlib rclcpp)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS fresh_range_layer LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_range_clear test/test_range_clear.cpp)
  target_compile_features(test_range_clear PUBLIC cxx_std_17)
  ament_target_dependencies(test_range_clear nav2_costmap_2d pluginlib rclcpp)
  add_test(NAME range_clear COMMAND test_range_clear)
endif()
ament_package()
