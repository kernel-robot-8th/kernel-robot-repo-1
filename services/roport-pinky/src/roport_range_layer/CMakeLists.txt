cmake_minimum_required(VERSION 3.8)
project(roport_range_layer)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(fresh_range_layer SHARED src/fresh_range_layer.cpp)
target_compile_features(fresh_range_layer PUBLIC cxx_std_17)
ament_target_dependencies(fresh_range_layer nav2_costmap_2d pluginlib rclcpp)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS fresh_range_layer LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
